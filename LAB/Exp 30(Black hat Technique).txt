import cv2
import numpy as np
from google.colab.patches import cv2_imshow

# Read the image (using the same image path as before)
img_path = '/content/134253236052803564.jpg'
img = cv2.imread(img_path)

# Check if image was loaded
if img is None:
    print(f"Error: Could not load image from {img_path}")
else:
    # Convert to grayscale
    gray_img = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)

    # Define a kernel (structuring element)
    kernel = np.ones((5, 5), np.uint8)

    # Apply the Black hat operation
    # Black hat = Closing of the image - Original Image
    blackhat_img = cv2.morphologyEx(gray_img, cv2.MORPH_BLACKHAT, kernel)

    # Display the original grayscale and the Black hat images
    print("Original Grayscale Image:")
    cv2_imshow(gray_img)
    print("Black hat Image:")
    cv2_imshow(blackhat_img)
