import cv2
import numpy as np
from google.colab.patches import cv2_imshow

# Read the image in grayscale
img_path = '/content/134253236052803564.jpg'
img = cv2.imread(img_path, cv2.IMREAD_GRAYSCALE)

# Perform Sobel edge detection along the X-axis (dx=1, dy=0)
# Using cv2.CV_64F prevents overflow from negative gradients
sobel_x = cv2.Sobel(img, cv2.CV_64F, 1, 0, ksize=3)

# Convert the output back to 8-bit unsigned integer (0-255)
sobel_x_abs = cv2.convertScaleAbs(sobel_x)

# Display results
print("Sobel Edge Detection (X-axis):")
cv2_imshow(sobel_x_abs)
