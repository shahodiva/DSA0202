import cv2
import numpy as np
from google.colab.patches import cv2_imshow

# Read the image
img_path = '/content/134253236052803564.jpg'
img = cv2.imread(img_path)

# Define Laplacian kernel with a negative center coefficient (-4)
# Standard 4-neighbor kernel with negative center:
laplacian_kernel = np.array([[ 0,  1,  0],
                             [ 1, -4,  1],
                             [ 0,  1,  0]], dtype=np.float32)

# Alternative 8-neighbor kernel with negative center (-8):
# laplacian_kernel = np.array([[ 1,  1,  1],
#                              [ 1, -8,  1],
#                              [ 1,  1,  1]], dtype=np.float32)

# Apply Laplacian filter (use CV_64F to avoid overflow)
laplacian = cv2.filter2D(img, cv2.CV_64F, laplacian_kernel)

# Sharpening formula for negative center: Original - Laplacian
sharpened = img.astype(np.float64) - laplacian

# Clip values to valid [0, 255] range and convert to 8-bit unsigned int
sharpened = np.clip(sharpened, 0, 255).astype(np.uint8)

# Display results
print("Sharpened Image (Negative Center Laplacian):")
cv2_imshow(sharpened)
