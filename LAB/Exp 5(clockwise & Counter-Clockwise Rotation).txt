import cv2
from google.colab.patches import cv2_imshow

# Read the image
img_path = '/content/134253236052803564.jpg'
img = cv2.imread(img_path)

# 1. Rotate 90 degrees Clockwise
clockwise_img = cv2.rotate(img, cv2.ROTATE_90_CLOCKWISE)

# 2. Rotate 90 degrees Counter-Clockwise
counter_clockwise_img = cv2.rotate(img, cv2.ROTATE_90_COUNTERCLOCKWISE)

# Display results
print("Clockwise Rotation (90°):")
cv2_imshow(clockwise_img)

print("Counter-Clockwise Rotation (90°):")
cv2_imshow(counter_clockwise_img)
