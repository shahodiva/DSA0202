import cv2
import numpy as np
from google.colab.patches import cv2_imshow

# Read the image
img_path = '/content/134253236052803564.jpg'
img = cv2.imread(img_path)

# Define Laplacian kernel with a positive center coefficient (+4)
# Standard 4-neighbor kernel with positive center:
laplacian_positive_center = np.array([
    [ 0, -1,  0],
    [-1,  4, -1],
    [ 0, -1,  0]
], dtype=np.float32)

# Alternative 8-neighbor kernel with positive center (+8):
# laplacian_positive_center = np.array([
#     [-1, -1, -1],
#     [-1,  8, -1],
#     [-1, -1, -1]
# ], dtype=np.float32)

# Compute Laplacian filter response (using CV_64F to prevent data overflow)
laplacian = cv2.filter2D(img, cv2.CV_64F, laplacian_positive_center)

# Sharpening formula for positive center: Original + Laplacian
sharpened = img.astype(np.float64) + laplacian

# Clip pixel values to [0, 255] range and convert to 8-bit unsigned integer
sharpened = np.clip(sharpened, 0, 255).astype(np.uint8)

# Display results
print("Sharpened Image (Positive Center Laplacian):")
cv2_imshow(sharpened)
