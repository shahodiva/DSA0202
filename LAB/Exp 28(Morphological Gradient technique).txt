import cv2
import numpy as np
from google.colab.patches import cv2_imshow

# Read the image (using the same image path as before)
img_path = '/content/134253236052803564.jpg'
img = cv2.imread(img_path)

# Check if image was loaded
if img is None:
    print(f"Error: Could not load image from {img_path}")
else:
    # Convert to grayscale and then to binary for morphological operations
    gray_img = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
    _, binary_img = cv2.threshold(gray_img, 127, 255, cv2.THRESH_BINARY) # Using THRESH_BINARY for consistency, or THRESH_BINARY_INV depending on object/background

    # Define a kernel (structuring element)
    kernel = np.ones((5, 5), np.uint8)

    # Apply the morphological gradient operation
    # Gradient = Dilation - Erosion
    gradient_img = cv2.morphologyEx(binary_img, cv2.MORPH_GRADIENT, kernel)

    # Display the original binary and the gradient images
    print("Original Binary Image:")
    cv2_imshow(binary_img)
    print("Morphological Gradient Image:")
    cv2_imshow(gradient_img)
