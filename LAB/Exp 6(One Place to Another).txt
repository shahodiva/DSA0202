import cv2
import numpy as np
from google.colab.patches import cv2_imshow

# Read the image
img_path = '/content/134253236052803564.jpg'
img = cv2.imread(img_path)
h, w = img.shape[:2]

# Define translation offsets: shift 100px Right (x), 50px Down (y)
tx, ty = 100, 50

# Create translation matrix
M = np.float32([[1, 0, tx],
                [0, 1, ty]])

# Move image to new position
translated_img = cv2.warpAffine(img, M, (w, h))

# Display result
print(f"Shifted Right by {tx}px and Down by {ty}px:")
cv2_imshow(translated_img)
