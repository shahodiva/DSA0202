import cv2
import numpy as np
from google.colab.patches import cv2_imshow

# Read the image (using the same image path as before)
img_path = '/content/134253236052803564.jpg'
img = cv2.imread(img_path)

# Check if image was loaded
if img is None:
    print(f"Error: Could not load image from {img_path}")
else:
    # Convert the image to grayscale
    gray_img = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)

    # Apply Canny edge detector
    # The parameters are: image, lower_threshold, upper_threshold
    # Edges with intensity gradient above upper_threshold are definitely edges.
    # Edges with intensity gradient below lower_threshold are definitely not edges.
    # Edges with intensity gradient between both thresholds are classified as edges if they are connected to 'definite' edges.
    edges = cv2.Canny(gray_img, 100, 200)

    # Display the result
    print("Image Boundaries (Canny Edge Detection):")
    cv2_imshow(edges)
