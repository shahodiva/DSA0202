import cv2
from google.colab.patches import cv2_imshow

# Read the image
img_path = '/content/134253236052803564.jpg'
img = cv2.imread(img_path)

# 1. Bigger size (e.g., 2x scale)
bigger_img = cv2.resize(img, (0, 0), fx=2.0, fy=2.0, interpolation=cv2.INTER_CUBIC)

# 2. Smaller size (e.g., 0.5x scale)
smaller_img = cv2.resize(img, (0, 0), fx=0.5, fy=0.5, interpolation=cv2.INTER_AREA)

# Display results
print("Bigger Size:")
cv2_imshow(bigger_img)

print("Smaller Size:")
cv2_imshow(smaller_img)
