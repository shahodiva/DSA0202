import cv2
import numpy as np
from google.colab.patches import cv2_imshow

# Read the image
img_path = '/content/134253236052803564.jpg'
img = cv2.imread(img_path)

# 1. Apply Gaussian Blur to smooth the image
blurred = cv2.GaussianBlur(img, (9, 9), 10.0)

# 2. Perform Unsharp Masking using cv2.addWeighted
# Formula: image * alpha + blurred * beta + gamma
# Here alpha = 1.5 (weight of original) and beta = -0.5 (subtraction of blurred)
unsharp_img = cv2.addWeighted(img, 1.5, blurred, -0.5, 0)

# Display results
print("Original vs Unsharp Masked Image:")
cv2_imshow(unsharp_img)
