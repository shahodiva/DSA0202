import cv2
from google.colab.patches import cv2_imshow

# Read the image in grayscale for optimal edge detection
img_path = '/content/134253236052803564.jpg'
img = cv2.imread(img_path, cv2.IMREAD_GRAYSCALE)

# Optional: Apply Gaussian Blur to reduce noise before detecting edges
blurred_img = cv2.GaussianBlur(img, (5, 5), 0)

# Perform Canny Edge Detection (threshold1 = lower bound, threshold2 = upper bound)
edges = cv2.Canny(blurred_img, threshold1=100, threshold2=200)

# Display results
print("Canny Edge Detection Result:")
cv2_imshow(edges)
